rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /wishes/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['n','m','t'])
                    && request.resource.data.n is string && request.resource.data.n.size() > 0 && request.resource.data.n.size() <= 40
                    && request.resource.data.m is string && request.resource.data.m.size() > 0 && request.resource.data.m.size() <= 300
                    && request.resource.data.t is int;
      allow update, delete: if false;
    }
  }
}
